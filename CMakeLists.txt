cmake_minimum_required(VERSION 3.22)
project(MOVOX VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
add_executable(dsp_tests tests/DspTests.cpp)
add_test(NAME vocal_dynamics COMMAND dsp_tests)
option(MOVOX_VALIDATE_HOST "Include offline VST3 host validation" OFF)
option(MOVOX_PLUGIN "Build native JUCE plugin" ON)
if(MOVOX_PLUGIN)
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 51a8a6d7aeae7326956d747737ccf1575e61e209)
 FetchContent_MakeAvailable(JUCE)
 set(formats VST3)
 if(APPLE)
  list(APPEND formats AU)
 endif()
 juce_add_plugin(MOVOX COMPANY_NAME "MO AUDIO" BUNDLE_ID "audio.mo.movox" PLUGIN_MANUFACTURER_CODE MoAu PLUGIN_CODE MoVx FORMATS ${formats} PRODUCT_NAME "MO VOX" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE VST3_CATEGORIES Dynamics)
 target_sources(MOVOX PRIVATE Source/Plugin.cpp)
 target_compile_definitions(MOVOX PUBLIC JUCE_PLUGINHOST_VST3=$<BOOL:${MOVOX_VALIDATE_HOST}> JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 add_executable(processor_checks tests/ProcessorChecks.cpp)
 target_link_libraries(processor_checks PRIVATE MOVOX)
 add_test(NAME native_processor COMMAND processor_checks)
 if(MOVOX_VALIDATE_HOST)
  add_dependencies(processor_checks MOVOX_VST3)
  add_test(NAME vst3_host COMMAND processor_checks "${CMAKE_CURRENT_BINARY_DIR}/MOVOX_artefacts/$<CONFIG>/VST3/MO VOX.vst3")
 endif()
 target_link_libraries(MOVOX PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
endif()
